cmake_minimum_required(VERSION 3.16)
project(ducktape VERSION 0.1.0 LANGUAGES C)

set(DUCKTAPE_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release" CACHE PATH "Directory holding the libraries cargo built for ducktape-abi")

find_library(DUCKTAPE_SHARED_LIBRARY NAMES ducktape_abi PATHS "${DUCKTAPE_LIB_DIR}" NO_DEFAULT_PATH REQUIRED)
find_library(DUCKTAPE_STATIC_LIBRARY NAMES libducktape_abi.a ducktape_abi.lib PATHS "${DUCKTAPE_LIB_DIR}" NO_DEFAULT_PATH)

if(WIN32)
  set(DUCKTAPE_SYSTEM_LIBS ws2_32 userenv bcrypt ntdll advapi32)
elseif(APPLE)
  set(DUCKTAPE_SYSTEM_LIBS "")
else()
  set(DUCKTAPE_SYSTEM_LIBS pthread dl m)
endif()

add_library(ducktape SHARED IMPORTED GLOBAL)
set_target_properties(ducktape PROPERTIES IMPORTED_LOCATION "${DUCKTAPE_SHARED_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(WIN32)
  set_target_properties(ducktape PROPERTIES IMPORTED_IMPLIB "${DUCKTAPE_LIB_DIR}/ducktape_abi.dll.lib")
endif()
add_library(ducktape::ducktape ALIAS ducktape)

if(DUCKTAPE_STATIC_LIBRARY)
  add_library(ducktape_static STATIC IMPORTED GLOBAL)
  set_target_properties(ducktape_static PROPERTIES IMPORTED_LOCATION "${DUCKTAPE_STATIC_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include" INTERFACE_LINK_LIBRARIES "${DUCKTAPE_SYSTEM_LIBS}")
  add_library(ducktape::ducktape_static ALIAS ducktape_static)
endif()

enable_testing()
add_executable(smoke test/smoke.c)
target_link_libraries(smoke PRIVATE ducktape::ducktape)
add_test(NAME smoke COMMAND smoke)
if(DUCKTAPE_STATIC_LIBRARY)
  add_executable(smoke_static test/smoke.c)
  target_link_libraries(smoke_static PRIVATE ducktape::ducktape_static)
  add_test(NAME smoke_static COMMAND smoke_static)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(DUCKTAPE_PKGCONFIG_PRIVATE "")
foreach(lib IN LISTS DUCKTAPE_SYSTEM_LIBS)
  string(APPEND DUCKTAPE_PKGCONFIG_PRIVATE " -l${lib}")
endforeach()
configure_file(ducktape.pc.in "${CMAKE_CURRENT_BINARY_DIR}/ducktape.pc" @ONLY)
configure_package_config_file(cmake/ducktapeConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ducktapeConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/ducktape" PATH_VARS CMAKE_INSTALL_INCLUDEDIR CMAKE_INSTALL_LIBDIR)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ducktapeConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
install(FILES include/ducktape.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES "${DUCKTAPE_SHARED_LIBRARY}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
if(DUCKTAPE_STATIC_LIBRARY)
  install(FILES "${DUCKTAPE_STATIC_LIBRARY}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
endif()
if(WIN32)
  install(FILES "${DUCKTAPE_LIB_DIR}/ducktape_abi.dll.lib" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
endif()
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ducktape.pc" DESTINATION "${CMAKE_INSTALL_LIBDIR}/pkgconfig")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ducktapeConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ducktapeConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/ducktape")
